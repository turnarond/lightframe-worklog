# 数据安全闭环 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 v2.11.0 实现本地写前快照、10+7 轮转、整份恢复、导入导出与 JSON 损坏保护。

**Architecture:** 新增 `src/backup.js` 承担恢复点生成/轮转/恢复/导入导出；`src/store.js` 在写盘前调用备份钩子并返回损坏状态；`main.js` 负责备份目录、IPC、恢复后暂停同步；渲染层在设置页提供备份列表与恢复/导入导出入口。

**Tech Stack:** Node 24、Electron 45、CommonJS、`node:test`、原生 `fs/promises`；不新增 npm 依赖。

**Spec:** `doc/03-设计/2026-11-21-数据安全闭环-设计.md`

## Global Constraints

- 版本目标：`2.11.0`；版本三件套在最后一笔提交同步。
- 数据格式：四个 `userData` JSON 文件不变；恢复只写 `tasks` / `dailyLogs` / `settings`，`syncmeta` 清空。
- 不新增 npm 依赖；时间戳统一毫秒；文档与提交信息中文。
- 所有数据写入仍使用 temp + rename；备份失败必须中止业务写入。
- 自动备份默认开启（`settings.backup.enabled = true`）；恢复后 `settings.syncPaused = true`。
- 恢复点保留：最近 10 份 + 最近 7 个自然日每天最后 1 份。
- 手动导出默认脱敏 `syncCode` / `serverUrl`；只有显式勾选才包含。
- 每个任务结束前必须跑相关测试并单独提交；不夹带无关改动。

`src/backup.js`：

```js
'use strict';
const fs = require('node:fs');
const fsp = fs.promises;
const path = require('node:path');
const crypto = require('node:crypto');

const DATA_FILES = ['tasks.json', 'dailyLogs.json', 'settings.json', 'syncmeta.json'];

function sha256(text) {
  return crypto.createHash('sha256').update(text).digest('hex');
}

function localDay(ts) {
  const d = new Date(ts);
  return d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0') +
    '-' + String(d.getDate()).padStart(2, '0');
}

function idFor(ts) {
  const d = new Date(ts);
  const p = (n, w = 2) => String(n).padStart(w, '0');
  return d.getFullYear() + p(d.getMonth() + 1) + p(d.getDate()) + '-' +
    p(d.getHours()) + p(d.getMinutes()) + p(d.getSeconds()) + '-' + p(d.getMilliseconds(), 3);
}

async function readIfExists(file) {
  try { return await fsp.readFile(file, 'utf8'); } catch (e) {
    if (e.code === 'ENOENT') return null;
    throw e;
  }
}

function createBackupManager(options) {
  const dataDir = options.dataDir;
  const backupDir = options.backupDir;
  const appVersion = options.appVersion || '';
  const recentLimit = options.recentLimit || 10;
  const dailyDays = options.dailyDays || 7;
  const now = options.now || Date.now;
  let queue = Promise.resolve();

  async function beforeWrite(reason) {
    const run = async () => {
      const contents = {};
      let any = false;
      for (const name of DATA_FILES) {
        const text = await readIfExists(path.join(dataDir, name));
        if (text !== null) any = true;
        contents[name] = text === null ? 'null' : text;
      }
      if (!any) return { created: false, skipped: 'empty' };
      const setHash = sha256(DATA_FILES.map((n) => sha256(contents[n])).join(''));
      const existing = await list();
      if (existing.length && existing[0].setHash === setHash) {
        return { created: false, setHash, skipped: 'unchanged' };
      }
      const ts = now();
      const id = idFor(ts);
      const tmpDir = path.join(backupDir, '.tmp-' + id);
      const finalDir = path.join(backupDir, id);
      await fsp.mkdir(tmpDir, { recursive: true });
      const files = {};
      for (const name of DATA_FILES) {
        await fsp.writeFile(path.join(tmpDir, name), contents[name], 'utf8');
        files[name] = { sha256: sha256(contents[name]), size: Buffer.byteLength(contents[name]) };
      }
      const meta = {
        id, createdAt: ts, reason: reason || 'write', appVersion, setHash, files,
        taskCount: countTasks(contents['tasks.json']),
        logDays: countLogs(contents['dailyLogs.json']),
        hasSyncCode: hasSyncCode(contents['settings.json'])
      };
      await fsp.writeFile(path.join(tmpDir, 'meta.json'), JSON.stringify(meta, null, 2), 'utf8');
      await fsp.rename(tmpDir, finalDir);
      await prune();
      return { created: true, id, setHash };
    };
    const result = queue.then(run, run);
    queue = result.then(() => {}, () => {});
    return result;
  }

  async function list() {
    let entries = [];
    try { entries = await fsp.readdir(backupDir); } catch (e) { return []; }
    const out = [];
    for (const name of entries) {
      if (name.startsWith('.')) continue;
      try {
        const meta = JSON.parse(await fsp.readFile(path.join(backupDir, name, 'meta.json'), 'utf8'));
        if (meta && meta.id === name) out.push(meta);
      } catch (e) { /* 未完成或损坏的恢复点忽略 */ }
    }
    return out.sort((a, b) => b.createdAt - a.createdAt);
  }

  async function prune() {
    const all = await list();
    const keep = new Set(all.slice(0, recentLimit).map((m) => m.id));
    const seen = new Set();
    for (const meta of all) {
      const day = localDay(meta.createdAt);
      if (seen.has(day) || seen.size >= dailyDays) continue;
      if (now() - meta.createdAt > dailyDays * 86400000) continue;
      seen.add(day);
      keep.add(meta.id);
    }
    for (const meta of all) {
      if (!keep.has(meta.id)) {
        try { await fsp.rm(path.join(backupDir, meta.id), { recursive: true, force: true }); }
        catch (e) { /* 下次轮转重试 */ }
      }
    }
  }

  async function spaceUsage() {
    let total = 0;
    for (const meta of await list()) {
      total += Buffer.byteLength(JSON.stringify(meta));
    }
    try {
      for (const name of await fsp.readdir(backupDir)) {
        const dir = path.join(backupDir, name);
        if (name.startsWith('.')) continue;
        for (const f of await fsp.readdir(dir)) {
          total += (await fsp.stat(path.join(dir, f))).size;
        }
      }
    } catch (e) { /* 目录尚未创建 */ }
    return total;
  }

  return { beforeWrite, list, prune, spaceUsage };
}

function countTasks(text) {
  try { const v = JSON.parse(text); return Array.isArray(v) ? v.length : 0; } catch (e) { return 0; }
}
function countLogs(text) {
  try { const v = JSON.parse(text); return v && typeof v === 'object' ? Object.keys(v).length : 0; } catch (e) { return 0; }
}
function hasSyncCode(text) {
  try { const v = JSON.parse(text); return !!(v && v.syncCode); } catch (e) { return false; }
}

module.exports = { DATA_FILES, createBackupManager, sha256, localDay };
```

```js
async function writeAtomic(file, text) {
  const tmp = file + '.tmp';
  await fsp.mkdir(path.dirname(file), { recursive: true });
  await fsp.writeFile(tmp, text, 'utf8');
  await fsp.rename(tmp, file);
}

function maskSettings(text, includeSecrets) {
  const s = JSON.parse(text);
  if (!includeSecrets) { s.syncCode = ''; s.serverUrl = ''; }
  return JSON.stringify(s, null, 2);
}

async function restoreSet(id) {
  const dir = path.join(backupDir, id);
  const pre = await beforeWrite('pre-restore');
  const restored = [];
  const originals = {};
  try {
    for (const name of ['tasks.json', 'dailyLogs.json', 'settings.json']) {
      const source = path.join(dir, name);
      const text = await fsp.readFile(source, 'utf8');
      const target = path.join(dataDir, name);
      originals[name] = await fsp.readFile(target, 'utf8').catch(() => null);
      await writeAtomic(target, text);
      restored.push(name);
    }
    return { preRestoreId: pre.id || '', restoredFiles: restored };
  } catch (err) {
    for (const name of restored) {
      const target = path.join(dataDir, name);
      if (originals[name] === null) await fsp.rm(target, { force: true });
      else await writeAtomic(target, originals[name]);
    }
    throw err;
  }
}

async function exportBundle(id, opts) {
  const includeSecrets = !!(opts && opts.includeSecrets);
  const dir = path.join(backupDir, id);
  const meta = JSON.parse(await fsp.readFile(path.join(dir, 'meta.json'), 'utf8'));
  const files = {
    tasks: JSON.parse(await fsp.readFile(path.join(dir, 'tasks.json'), 'utf8')),
    dailyLogs: JSON.parse(await fsp.readFile(path.join(dir, 'dailyLogs.json'), 'utf8')),
    settings: JSON.parse(maskSettings(await fsp.readFile(path.join(dir, 'settings.json'), 'utf8'), includeSecrets)),
    syncmeta: JSON.parse(await fsp.readFile(path.join(dir, 'syncmeta.json'), 'utf8'))
  };
  const checksums = {};
  for (const key of Object.keys(files)) checksums[key] = sha256(JSON.stringify(files[key]));
  const bundle = {
    format: 'lightframe-backup', formatVersion: 1, exportedAt: new Date(now()).toISOString(),
    appVersion, files, checksums
  };
  return { text: JSON.stringify(bundle, null, 2), filename: 'lightframe-backup-' + meta.id + '.json' };
}

async function importBundle(text) {
  const bundle = JSON.parse(text);
  if (!bundle || bundle.format !== 'lightframe-backup' || bundle.formatVersion !== 1) {
    throw new Error('不是受支持的备份文件');
  }
  for (const key of ['tasks', 'dailyLogs', 'settings', 'syncmeta']) {
    if (!(key in bundle.files)) throw new Error('备份缺少 ' + key);
    if (checksumOf(bundle.files[key]) !== bundle.checksums[key]) throw new Error(key + ' 校验失败');
  }
  return {
    bundle,
    summary: {
      taskCount: Array.isArray(bundle.files.tasks) ? bundle.files.tasks.length : 0,
      logDays: Object.keys(bundle.files.dailyLogs || {}).length,
      hasSyncCode: !!(bundle.files.settings && bundle.files.settings.syncCode)
    }
  };
}

async function restoreBundle(bundle) {
  const pre = await beforeWrite('pre-restore');
  const writes = [
    ['tasks.json', bundle.files.tasks],
    ['dailyLogs.json', bundle.files.dailyLogs],
    ['settings.json', bundle.files.settings]
  ];
  const originals = {};
  const done = [];
  try {
    for (const [name, value] of writes) {
      const target = path.join(dataDir, name);
      originals[name] = await fsp.readFile(target, 'utf8').catch(() => null);
      await writeAtomic(target, JSON.stringify(value, null, 2));
      done.push(name);
    }
    return { preRestoreId: pre.id || '', restoredFiles: done };
  } catch (err) {
    for (const name of done) {
      const target = path.join(dataDir, name);
      if (originals[name] === null) await fsp.rm(target, { force: true });
      else await writeAtomic(target, originals[name]);
    }
    throw err;
  }
}

function checksumOf(value) {
  return sha256(JSON.stringify(value));
}
```

async function readJsonWithStatus(filePath) {
  try {
    const text = await fsp.readFile(filePath, 'utf8');
    return { data: JSON.parse(text), missing: false, corrupt: false, error: '' };
  } catch (err) {
    if (err.code === 'ENOENT') return { data: undefined, missing: true, corrupt: false, error: '' };
    return { data: undefined, missing: false, corrupt: true, error: err.message };
  }
}

function withHook(options) {
  const fn = options && typeof options.beforeWrite === 'function' ? options.beforeWrite : null;
  return fn ? () => fn() : async () => {};
}
