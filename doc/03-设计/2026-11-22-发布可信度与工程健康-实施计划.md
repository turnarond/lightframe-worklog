// scripts/release-lib.js
'use strict';
const fs = require('node:fs');
const crypto = require('node:crypto');

function parseReleaseArgs(argv) {
  const out = { _: [] };
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i];
    if (!a.startsWith('--')) { out._.push(a); continue; }
    const eq = a.indexOf('=');
    if (eq >= 0) { out[a.slice(2, eq)] = a.slice(eq + 1); continue; }
    const key = a.slice(2);
    if (argv[i + 1] && !argv[i + 1].startsWith('--')) { out[key] = argv[++i]; }
    else out[key] = true;
  }
  return out;
}

function validateVersionTrio(v, expected) {
  const errs = [];
  if (v.packageVersion !== expected) errs.push('package.json version 不等于 ' + expected);
  if (String(v.versionText).trim() !== expected) errs.push('VERSION 不等于 ' + expected);
  if (!v.changelog.includes('## [' + expected + ']')) errs.push('CHANGELOG 缺少 ' + expected + ' 段');
  return errs;
}

function isCleanStatus(text) { return text.trim() === ''; }

function hashFile(file, algo) {
  return crypto.createHash(algo).update(fs.readFileSync(file)).digest('hex');
}
function sha256File(f) { return hashFile(f, 'sha256'); }
function sha512File(f) { return hashFile(f, 'sha512'); }
function md5File(f) { return hashFile(f, 'md5'); }

function renderReleaseNotes({ version, changeSummary, assets }) {
  const lines = ['## 变更', changeSummary || '- 发布可信度与门禁收口。', '', '## 资产'];
  for (const a of assets || []) lines.push('- `' + a.name + '` — ' + a.size + ' 字节 — sha256 `' + a.sha256 + '`');
  return '# v' + version + '\n\n' + lines.join('\n') + '\n';
}

function releaseCommands(version, assets) {
  const files = assets && assets.length ? assets : [
    'dist/lightframe-worklog-setup-' + version + '.exe',
    'dist/lightframe-worklog-setup-' + version + '.exe.blockmap',
    'dist/latest.yml'
  ];
  return [
    'git push origin main',
    'git tag -a v' + version + ' -m "v' + version + ' · 2026-11-22"',
    'git push origin v' + version,
    'gh release create v' + version + ' ' + files.join(' ') + ' --notes-file dist/release-notes-' + version + '.md'
  ];
}

module.exports = { parseReleaseArgs, validateVersionTrio, isCleanStatus, renderReleaseNotes, releaseCommands, sha256File, sha512File, md5File };
```

// scripts/release-lib.js 追加
function runCheck(input) {
  const errors = [];
  if (!isCleanStatus(input.statusText || '')) errors.push('工作区不干净');
  errors.push(...validateVersionTrio({
    packageVersion: input.packageVersion,
    versionText: fs.readFileSync(require('node:path').join(input.cwd, 'VERSION'), 'utf8'),
    changelog: input.changelog || ''
  }, input.version));
  if (input.readmeText && !input.readmeText.includes('当前版本：**v' + input.version + '**')) {
    errors.push('README 当前版本不是 v' + input.version);
  }
  if (!input.allowExistingTag && (input.tags || []).includes('v' + input.version)) {
    errors.push('tag v' + input.version + ' 已存在');
  }
  return { ok: errors.length === 0, errors };
}
module.exports.runCheck = runCheck;

const { execFileSync } = require('node:child_process');

function git(args) { return execFileSync('git', args, { encoding: 'utf8' }).trim(); }
function preflight(opts) {
  const version = opts.version || require('../package.json').version;
  const result = lib.runCheck({
    cwd: process.cwd(),
    version,
    packageVersion: require('../package.json').version,
    changelog: fs.readFileSync('CHANGELOG.md', 'utf8'),
    readmeText: fs.readFileSync('README.md', 'utf8'),

    tags: git(['tag', '--list']).split(/\r?\n/).filter(Boolean),
    allowExistingTag: !!opts.check || !!opts['allow-existing-tag']
  });
  if (!result.ok) { console.error(result.errors.join('\n')); process.exit(1); }
  // 发布文档检查仅在完整模式执行；--check 只做代码/版本预检

  return version;
}

// scripts/release-lib.js 追加
function findCachedElectron(version, cacheRoot) {
  const fs = require('node:fs');
  const path = require('node:path');
  const target = 'electron-v' + version + '-win32-x64.zip';
  if (!fs.existsSync(cacheRoot)) return null;
  const dirs = fs.readdirSync(cacheRoot, { withFileTypes: true }).filter((d) => d.isDirectory());
  for (const d of dirs) {
    const hit = path.join(cacheRoot, d.name, target);
    if (fs.existsSync(hit)) return hit;
  }
  return null;
}
module.exports.findCachedElectron = findCachedElectron;

const fs = require('node:fs');
const path = require('node:path');
const { execFileSync } = require('node:child_process');
const { findCachedElectron } = require('./release-lib.js');

function ensureElectron({ version, distDir, cacheRoot }) {
  const versionFile = path.join(distDir, 'version');
  if (fs.existsSync(versionFile) && fs.readFileSync(versionFile, 'utf8').trim() === version) return;
  const zip = findCachedElectron(version, cacheRoot);
  if (!zip) throw new Error('未找到 Electron 缓存；请联网安装或传入 --electron-dist');
  fs.mkdirSync(distDir, { recursive: true });
  execFileSync('powershell', ['-NoProfile', '-Command',
    'Expand-Archive -LiteralPath "' + zip + '" -DestinationPath "' + distDir + '" -Force'], { stdio: 'inherit' });
  fs.writeFileSync(path.join(distDir, '..', 'path.txt'), 'electron.exe');
}
module.exports = { ensureElectron };

// tools/verify-update-lib.js
'use strict';
const fs = require('node:fs');
const os = require('node:os');
const path = require('node:path');
const crypto = require('node:crypto');

function renderAppUpdateYml({ url, cacheName }) {
  return 'provider: generic\nurl: ' + url + '\nupdaterCacheDirName: ' + cacheName + '\n';
}

function installerArgs(testRoot) {
  return ['/S', '/D=' + path.join(testRoot, 'app')];
}

function assertSafeTestRoot({ testRoot, dailyInstallPath, dailyUserDataPath }) {
  const errs = [];
  const root = path.resolve(testRoot);
  const tmp = path.resolve(os.tmpdir());
  if (!root.startsWith(tmp + path.sep)) errs.push('test-root 必须位于临时目录');
  for (const p of [dailyInstallPath, dailyUserDataPath]) {
    const r = path.resolve(p);
    if (root === r || root.startsWith(r + path.sep)) errs.push('拒绝触碰日常安装/数据: ' + r);
  }
  return errs;
}

function findDownloadedInstaller(cacheDir, sha256) {
  if (!fs.existsSync(cacheDir)) return null;
  const stack = [cacheDir];
  while (stack.length) {
    const dir = stack.pop();
    for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
      const p = path.join(dir, e.name);
      if (e.isDirectory()) stack.push(p);
      else if (/\.exe$/i.test(e.name)) {
        const h = crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');
        if (h === sha256) return p;
      }
    }
  }
  return null;
}

module.exports = { renderAppUpdateYml, installerArgs, assertSafeTestRoot, findDownloadedInstaller };
```

```js
// tools/verify-update.js 关键片段
const { spawn, spawnSync } = require('node:child_process');
const http = require('node:http');
const fs = require('node:fs');
const os = require('node:os');
const path = require('node:path');
const lib = require('./verify-update-lib.js');

function installOld(exe, testRoot) {
  const r = spawnSync(exe, lib.installerArgs(testRoot), { stdio: 'inherit' });
  if (r.status !== 0) throw new Error('旧版安装失败: ' + r.status);
  const app = path.join(testRoot, 'app', '光影工作志.exe');
  if (!fs.existsSync(app)) throw new Error('未找到旧版 exe: ' + app);
  return app;
}

function patchFeed(appDir, url) {
  const yml = path.join(appDir, 'resources', 'app-update.yml');
  if (!fs.existsSync(yml)) throw new Error('缺少 app-update.yml: ' + yml);
  fs.writeFileSync(yml, lib.renderAppUpdateYml({ url, cacheName: 'lw-update-e2e' }));
  return yml;
}

function startServer(distDir, port) {
  const server = http.createServer((req, res) => {
    const p = path.join(distDir, decodeURIComponent(req.url.split('?')[0]));
    if (!fs.existsSync(p) || !fs.statSync(p).isFile()) { res.statusCode = 404; res.end(); return; }
    res.setHeader('Content-Type', 'application/octet-stream');
    fs.createReadStream(p).pipe(res);
  });
  return new Promise((resolve) => server.listen(port, '127.0.0.1', () => resolve(server)));
}
```
